using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.SceneManagement;using System.Linq;using System.IO;using System;
internal class CommandScript:IRunCommand{public void Execute(ExecutionResult r){if(File.Exists("Captures/Aftermath/Phase3_20261013/refinement.txt"))throw new Exception("Refinement already applied; preserve final meshes.");var root=GameObject.Find("Aftermath - Phase 3 damaged homes");var ground=Object.FindObjectsByType<Collider>(FindObjectsSortMode.None).Where(c=>c.name.StartsWith("Measured terrain")).ToArray();var old=GameObject.Find("Aftermath - damaged homes and yards");int n=0,flaps=0;
foreach(var f in root.GetComponentsInChildren<MeshFilter>()){var m=f.sharedMesh;var vs=m.vertices;bool change=false;
if(f.name=="Weathered pale fence"){for(int i=0;i+71<vs.Length;i+=72){var block=vs.Skip(i).Take(72).ToArray();var b=new Bounds(block[0],Vector3.zero);foreach(var v in block)b.Encapsulate(v);if(b.size.y<.18f||b.size.y<Mathf.Max(b.size.x,b.size.z))continue;float support=float.NegativeInfinity;foreach(var c in ground)if(c.Raycast(new Ray(new Vector3(b.center.x,25,b.center.z),Vector3.down),out var h,60))support=Mathf.Max(support,h.point.y);for(int j=i;j<i+72;j++)vs[j].y=b.max.y-vs[j].y+support+.01f;n++;change=true;}}
if(f.name.EndsWith("displaced roof metal")){var name=f.transform.parent.name.Split(new[]{" - "},StringSplitOptions.None)[0];var beam=old.GetComponentsInChildren<Transform>().Where(t=>t.name==name+" exposed roof rafter").OrderByDescending(t=>t.position.y).FirstOrDefault();if(beam){for(int i=vs.Length-864;i<vs.Length;i++)vs[i]-=beam.up*.085f;flaps++;change=true;}}
if(change){r.RegisterObjectModification(m);m.vertices=vs;m.RecalculateNormals();m.RecalculateBounds();EditorUtility.SetDirty(m);}
}AssetDatabase.SaveAssets();EditorSceneManager.SaveScene(SceneManager.GetActiveScene());var msg="Refined "+n+" standing fence boards with upper splinters and ground contact; seated "+flaps+" roof strips onto existing rafters.";File.WriteAllText("Captures/Aftermath/Phase3_20261013/refinement.txt",msg);r.Log(msg);}}
