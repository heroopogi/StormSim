 float Rand(float a,float b){return Mathf.Lerp(a,b,(float)rng.NextDouble());}
 Batch NewBatch(string name,Color color,float smooth,float mode,int queue){
  var path=Folder+"/"+name+".mat";var m=AssetDatabase.LoadAssetAtPath<Material>(path);if(!m){m=new Material(Shader.Find("StormSim/Aftermath Flood Residue"));AssetDatabase.CreateAsset(m,path);}else result.RegisterObjectModification(m);m.name=name;m.SetColor("_BaseColor",color);m.SetFloat("_Smoothness",smooth);m.SetFloat("_Mode",mode);m.SetFloat("_WaterLevel",FloodLevel);m.renderQueue=queue;
  EditorUtility.SetDirty(m);return new Batch{name=name,mat=m};
 }
 Vector3 Surface(float x,float z){
  string key=x.ToString("F3",System.Globalization.CultureInfo.InvariantCulture)+","+z.ToString("F3",System.Globalization.CultureInfo.InvariantCulture);
  if(surfaces.TryGetValue(key,out var cached))return cached;
  float y=float.NegativeInfinity;var normal=Vector3.up;
  var ray=new Ray(new Vector3(x,25,z),Vector3.down);
  foreach(var c in terrain)if(c.Raycast(ray,out var h,60)&&h.point.y>y)y=h.point.y;
  if(!float.IsFinite(y))return new Vector3(x,float.NaN,z);
  var hits=Physics.RaycastAll(ray,60,~0,QueryTriggerInteraction.Ignore);
  foreach(var h in hits)if(h.normal.y>.75f&&(h.collider.name=="Felix Amante Drive"||h.collider.name=="Main entrance to quadrangle"||h.collider.name.StartsWith("service "))&&h.point.y>y&&h.point.y-y<1.2f)y=h.point.y;
  foreach(var h in hits){
   if(h.collider is CharacterController||h.collider.GetComponentInParent<SibucaoResident>()||h.collider.GetComponentInParent<SibucaoPlayer>())continue;
   if(h.point.y>y+.16f){y=float.NaN;break;}
  }
  var value=new Vector3(x,y,z);surfaces[key]=value;return value;
 }
 void Patch(Batch b,float x,float z,float width,float length,float yaw,float level){
  float step=b==pools?.14f:.32f;int nx=Mathf.CeilToInt(width/step)+1,nz=Mathf.CeilToInt(length/step)+1;
  int start=b.v.Count;var rot=Quaternion.Euler(0,yaw,0);var valid=new bool[nx*nz];
  for(int iz=0;iz<nz;iz++)for(int ix=0;ix<nx;ix++){
   float u=(float)ix/(nx-1)*2-1,v=(float)iz/(nz-1)*2-1;
   var q=new Vector3(x,0,z)+rot*new Vector3(u*width*.5f,0,v*length*.5f);
   float radius=Mathf.Sqrt(u*u+v*v);float wobble=(Mathf.PerlinNoise(q.x*1.9f+31,q.z*1.9f)-.5f)*.23f;
   float alpha=Mathf.SmoothStep(0,1,Mathf.Clamp01((1-radius+wobble)/.48f));
   var p=Surface(q.x,q.z);bool ok=float.IsFinite(p.y);
   if(ok&&b==pools){float depth=level-p.y;ok=depth>=.001f&&depth<=.04f;alpha*=Mathf.SmoothStep(0,1,depth/.012f);p.y=level+.004f;}
   else if(ok)p.y+=b==silt?.009f:.014f;
   if(!ok){p=new Vector3(q.x,0,q.z);alpha=0;}
   valid[iz*nx+ix]=ok;b.v.Add(p);b.c.Add(new Color(1,1,1,alpha));
  }
  for(int iz=0;iz<nz-1;iz++)for(int ix=0;ix<nx-1;ix++){
   int a=iz*nx+ix,bb=a+1,c=a+nx,d=c+1;
   Tri(b,start,a,c,bb,valid);Tri(b,start,bb,c,d,valid);
  }
 }
 void Tri(Batch b,int start,int a,int c,int d,bool[] valid){if(valid[a]&&valid[c]&&valid[d]&&Mathf.Max(b.v[start+a].y,Mathf.Max(b.v[start+c].y,b.v[start+d].y))-Mathf.Min(b.v[start+a].y,Mathf.Min(b.v[start+c].y,b.v[start+d].y))<.09f){b.t.Add(start+a);b.t.Add(start+c);b.t.Add(start+d);}}
 float RouteDistance(Vector3 p){var center=session.route.Sample(session.route.Progress(p));return new Vector2(p.x-center.x,p.z-center.z).magnitude;}
 void Scatter(float x,float z,float yaw,int count){
  for(int i=0;i<count;i++){
   var q=Surface(x+Rand(-1.1f,1.1f),z+Rand(-1.0f,1));if(!float.IsFinite(q.y)||RouteDistance(q)<1.7f)continue;
   bool scrap=i%11==0;var b=scrap?paper:leaves;float angle=Rand(0,360)*Mathf.Deg2Rad,len=Rand(.09f,.23f),wide=len*Rand(.3f,.55f);
   var along=new Vector3(Mathf.Sin(angle),0,Mathf.Cos(angle));var across=Vector3.Cross(Vector3.up,along);
   var shape=scrap?new[]{-along*len-across*wide,along*len-across*wide,along*len+across*wide,-along*len+across*wide}:new[]{-along*len,-along*len*.25f-across*wide,along*len,along*len*.25f+across*wide};
   int at=b.v.Count;bool ok=true;
   foreach(var o in shape){var p=Surface(q.x+o.x,q.z+o.z);if(!float.IsFinite(p.y)){ok=false;p=q;}p.y+=.017f;b.v.Add(p);float tint=Rand(.7f,1.15f);b.c.Add(new Color(tint,tint,scrap?tint:tint*.8f,1));}
   if(ok){b.t.AddRange(new[]{at,at+1,at+2,at,at+2,at+3});}
  }
 }
 void WallTrace(MeshFilter f){
  var vs=f.sharedMesh.vertices;var ts=f.sharedMesh.triangles;
  for(int i=0;i<ts.Length;i+=3){
   var a=f.transform.TransformPoint(vs[ts[i]]);var b=f.transform.TransformPoint(vs[ts[i+1]]);var c=f.transform.TransformPoint(vs[ts[i+2]]);
   var n=Vector3.Cross(b-a,c-a).normalized;if(Mathf.Abs(n.y)>.35f)continue;
   var center=(a+b+c)/3;if(center.x<-58||center.x>18||center.z<642||center.z>696)continue;
   if(Mathf.Min(a.y,Mathf.Min(b.y,c.y))>FloodLevel+.12f)continue;
   var poly=new List<Vector3>{a,b,c};var clipped=new List<Vector3>();
   for(int k=0;k<poly.Count;k++){var p=poly[k];var next=poly[(k+1)%poly.Count];bool inside=p.y<=FloodLevel+.12f,ni=next.y<=FloodLevel+.12f;if(inside)clipped.Add(p);if(inside!=ni)clipped.Add(Vector3.Lerp(p,next,(FloodLevel+.12f-p.y)/(next.y-p.y)));}
   if(clipped.Count<3)continue;int at=stains.v.Count;foreach(var p in clipped){stains.v.Add(p+n*.006f);stains.c.Add(Color.white);}for(int k=1;k<clipped.Count-1;k++)stains.t.AddRange(new[]{at,at+k,at+k+1});
  }
 }
 void Save(Batch b){
  if(b.t.Count==0){notes.Add(b.name+": no accepted triangles");return;}
  var mesh=new UnityEngine.Mesh{name=b.name,indexFormat=IndexFormat.UInt32};mesh.SetVertices(b.v);mesh.SetColors(b.c);mesh.SetTriangles(b.t,0);mesh.RecalculateNormals();mesh.RecalculateBounds();
  var path=Folder+"/"+b.name+".asset";var saved=AssetDatabase.LoadAssetAtPath<UnityEngine.Mesh>(path);if(saved){result.RegisterObjectModification(saved);EditorUtility.CopySerialized(mesh,saved);result.DestroyObject(mesh);mesh=saved;EditorUtility.SetDirty(saved);}else AssetDatabase.CreateAsset(mesh,path);
  var child=root.Find(b.name);var g=child?child.gameObject:new GameObject(b.name);if(!child){result.RegisterObjectCreation(g);g.transform.SetParent(root,false);g.AddComponent<MeshFilter>();g.AddComponent<MeshRenderer>();}else result.RegisterObjectModification(g);g.GetComponent<MeshFilter>().sharedMesh=mesh;var mr=g.GetComponent<MeshRenderer>();mr.sharedMaterial=b.mat;mr.shadowCastingMode=ShadowCastingMode.Off;mr.receiveShadows=true;
  notes.Add(b.name+": "+b.t.Count/3+" triangles; "+b.v.Count+" vertices.");
 }
