bool Stand(Vector3 requested,out Vector3 position)
{
    position=requested;
    var ground=Physics.RaycastAll(requested+Vector3.up*.65f,Vector3.down,1.5f,~0,QueryTriggerInteraction.Ignore).Where(h=>!IsPlayer(h.collider)&&h.normal.y>.7f).OrderBy(h=>h.distance).ToArray();
    if(ground.Length==0)return false;position=ground[0].point+Vector3.up*.025f;
    return !Physics.OverlapCapsule(position+Vector3.up*.38f,position+Vector3.up*1.55f,.29f,~0,QueryTriggerInteraction.Ignore).Any(c=>!IsPlayer(c));
}

    var queue=new Queue<Vector2Int>();var parent=new Dictionary<Vector2Int,Vector2Int>();var positions=new Dictionary<Vector2Int,Vector3>{{a,start}};var seen=new HashSet<Vector2Int>{a};queue.Enqueue(a);
    while(queue.Count>0){var cur=queue.Dequeue();if(cur==b){var path=new List<Vector3>{goal};while(cur!=a){path.Add(positions[cur]);cur=parent[cur];}path.Reverse();return path.ToArray();}foreach(var offset in new[]{Vector2Int.up,Vector2Int.right,Vector2Int.down,Vector2Int.left}){var next=cur+offset;if(Mathf.Abs(next.x)>14||Mathf.Abs(next.y)>14||!seen.Add(next))continue;var p=origin+new Vector3(next.x*cell,0,next.y*cell);if(!Stand(p,out p)||Mathf.Abs(p.y-positions[cur].y)>.45f||!Clear(positions[cur],p))continue;positions[next]=p;parent[next]=cur;queue.Enqueue(next);}}
