 float RouteGap(Vector3 p){var q=session.route.Sample(session.route.Progress(p));return new Vector2(p.x-q.x,p.z-q.z).magnitude;}

 void Interior(Transform house,int severity){
 var vs=house.GetComponent<MeshFilter>().sharedMesh.vertices.Select(v=>new Vector2(v.x,v.z)).Distinct().OrderBy(v=>v.x).ThenBy(v=>v.y).ToList();

 foreach(var p in vs){while(hull.Count>=2&&cross(hull[hull.Count-2],hull[hull.Count-1],p)<=0)hull.RemoveAt(hull.Count-1);hull.Add(p);}int lower=hull.Count;
 for(int i=vs.Count-2;i>=0;i--){var p=vs[i];while(hull.Count>lower&&cross(hull[hull.Count-2],hull[hull.Count-1],p)<=0)hull.RemoveAt(hull.Count-1);hull.Add(p);}hull.RemoveAt(hull.Count-1);
 var points=hull.Select(v=>house.position+new Vector3(v.x*.90f,0,v.y*.90f)).ToArray();float y=points.Select(p=>Ground(p)).Max()+.035f;for(int i=0;i<points.Length;i++)points[i].y=y;

 }
