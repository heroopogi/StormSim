float Ground(Vector3 p){float y=float.NegativeInfinity;foreach(var c in ground)if(c.Raycast(new Ray(new Vector3(p.x,40,p.z),Vector3.down),out var hit,80))y=Mathf.Max(y,hit.point.y);if(!float.IsFinite(y))throw new Exception("No ground "+p);return y;}

var red=Mat("Recovery restricted red",new Color(.6f,.13f,.09f));

void Beam(string name,Vector3 a,Vector3 b,float width,Material material){var delta=b-a;if(delta.sqrMagnitude<.02f)return;Box(name,(a+b)*.5f,new Vector3(width,width,delta.magnitude),material,Quaternion.LookRotation(delta));}
int FindHouse(Vector3 p){int best=-1;float distance=float.MaxValue;for(int i=0;i<houses.Length;i++){var b=bounds[i];if(p.x<b.min.x+.1f||p.x>b.max.x-.1f||p.z<b.min.z+.1f||p.z>b.max.z-.1f)continue;float d=(p.x-b.center.x)*(p.x-b.center.x)+(p.z-b.center.z)*(p.z-b.center.z);if(d<distance){best=i;distance=d;}}return best;}
bool Hole(Vector3 p,int h){if(severity[h]==0)return false;var b=bounds[h];float dx=(p.x-b.center.x)/b.extents.x,dz=(p.z-b.center.z)/b.extents.z;float r=severity[h]==2?.84f:.58f;return (dx-.2f)*(dx-.2f)+(dz+.15f)*(dz+.15f)<r*r;}
var affected=new int[houses.Length];int meshCount=0,stained=0;
var sources=UnityEngine.Object.FindObjectsByType<MeshFilter>(FindObjectsSortMode.None).Where(f=>f.transform.parent&&(f.transform.parent.name=="Neighbourhood -1_6"||f.transform.parent.name=="Neighbourhood -1_7")&&f.name=="metal").OrderBy(f=>f.transform.parent.name+"/"+f.name).ToArray();
foreach(var f in sources)
{
    var source=f.sharedMesh;if(!source.isReadable||source.subMeshCount!=1)throw new Exception("Unexpected mesh layout "+f.name);
    var vertices=source.vertices.ToList();var normals=source.normals.ToList();var uv=source.uv.ToList();var colors=source.colors.ToList();
    while(normals.Count<vertices.Count)normals.Add(Vector3.up);while(uv.Count<vertices.Count)uv.Add(Vector2.zero);while(colors.Count<vertices.Count)colors.Add(Color.white);
    var output=new List<int>();bool changed=false;var rafters=new HashSet<int>();bool roof=f.name=="metal";
    int Mid(int a,int b){int n=vertices.Count;vertices.Add((vertices[a]+vertices[b])*.5f);normals.Add((normals[a]+normals[b]).normalized);uv.Add((uv[a]+uv[b])*.5f);colors.Add((colors[a]+colors[b])*.5f);return n;}
    void Tri(int a,int b,int c,int depth,int house)
    {
        if(depth>0){int ab=Mid(a,b),bc=Mid(b,c),ca=Mid(c,a);Tri(a,ab,ca,depth-1,house);Tri(ab,b,bc,depth-1,house);Tri(ca,bc,c,depth-1,house);Tri(ab,bc,ca,depth-1,house);return;}
        var p=f.transform.TransformPoint((vertices[a]+vertices[b]+vertices[c])/3);
        if(roof&&Hole(p,house)){affected[house]++;changed=true;return;}
        if(!roof)
        {
            foreach(int id in new[]{a,b,c})
            {
                var wp=f.transform.TransformPoint(vertices[id]);float height=wp.y-houses[house].transform.position.y;
                if(height<1.15f+.12f*Mathf.Sin(wp.x*2+wp.z*3))
                {
                    // Set from the source-like facade tint once; never compound shared-vertex tinting.
                    var col=colors[id];if(col.a>.99f){col=Color.Lerp(col,new Color(.22f,.17f,.105f),.72f);col.a=.98f;colors[id]=col;stained++;}
                }
            }
            changed=true;
        }
        output.Add(a);output.Add(b);output.Add(c);
    }
    var indices=source.triangles;
    for(int k=0;k<indices.Length;k+=3)
    {
        int a=indices[k],b=indices[k+1],c=indices[k+2];var p=f.transform.TransformPoint((vertices[a]+vertices[b]+vertices[c])/3);int h=FindHouse(p);
        bool eligible=h>=0&&(!roof||p.y>houses[h].transform.position.y+houses[h].wallHeightMeters-.6f);
        if(!eligible){output.Add(a);output.Add(b);output.Add(c);continue;}
        if(roof&&severity[h]>0)
        {
            var wa=f.transform.TransformPoint(vertices[a]);var wb=f.transform.TransformPoint(vertices[b]);var wc=f.transform.TransformPoint(vertices[c]);
            // Rafters follow the actual pitched roof plane, beneath the missing sheets.
            var normal=Vector3.Cross(wb-wa,wc-wa).normalized;
            if(Mathf.Abs(normal.y)>.2f&&(wb-wa).magnitude>2)
                for(int j=1;j<=5;j++)
                {
                    var ra=Vector3.Lerp(wa,wb,j/6f)-Vector3.up*.16f;var rb=Vector3.Lerp(wa,wc,j/6f)-Vector3.up*.16f;
                    if(Hole((ra+rb)*.5f,h))Beam(names[h]+" exposed roof rafter",ra,rb,.10f,timber);
                }
        }
        Tri(a,b,c,roof?3:2,h);
    }
    if(!changed)continue;
    var copy=UnityEngine.Object.Instantiate(source);copy.name=f.transform.parent.name+" aftermath "+f.name;copy.indexFormat=UnityEngine.Rendering.IndexFormat.UInt32;
    copy.Clear();copy.SetVertices(vertices);copy.SetNormals(normals);copy.SetUVs(0,uv);copy.SetColors(colors);copy.SetTriangles(output,0);copy.RecalculateBounds();
    var meshPath=folder+"/Damaged neighborhood "+meshCount+".asset";var savedMesh=AssetDatabase.LoadAssetAtPath<Mesh>(meshPath);
    if(savedMesh){EditorUtility.CopySerialized(copy,savedMesh);UnityEngine.Object.DestroyImmediate(copy);copy=savedMesh;}else AssetDatabase.CreateAsset(copy,meshPath);f.sharedMesh=copy;
    var collider=f.GetComponent<MeshCollider>();if(collider)collider.sharedMesh=copy;
    meshCount++;
}
for(int i=0;i<houses.Length;i++)
{
    var p=houses[i].transform.position;float progress=session.route.Progress(p);var road=session.route.Sample(progress);var toward=road-p;toward.y=0;toward.Normalize();var side=Vector3.Cross(Vector3.up,toward);
    var yard=p+toward*(Mathf.Min(bounds[i].extents.x,bounds[i].extents.z)+.65f);
    // Yard details stay back from the authored walking route.
    if(Vector3.Distance(new Vector3(yard.x,0,yard.z),new Vector3(road.x,0,road.z))<2.7f)yard-=toward*1.5f;
    yard.y=Ground(yard);
    for(int j=0;j<6;j++)
    {
        var foot=yard+side*(j*.32f-.8f);foot.y=Ground(foot);
        var rotation=Quaternion.LookRotation(toward)*Quaternion.Euler(j%3==0?65:8,0,j%2==0?-18:12);

    }
    if(severity[i]>0)
    {
        var sheet=yard+side*1.1f-toward*.5f;sheet.y=Ground(sheet)+.12f;

    }
    if(severity[i]==2)
    {
        var barrier=yard-toward*.25f;barrier.y=Ground(barrier)+.85f;

    }
}
