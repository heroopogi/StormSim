using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.SceneManagement;using StormSim.Sibucao;using System;using System.IO;using System.Linq;using System.Collections.Generic;
internal class CommandScript:IRunCommand{const string Dir="Captures/Aftermath/Phase3_20261013";public void Execute(ExecutionResult result){
if(EditorApplication.isPlaying||SceneManager.GetActiveScene().path!="Assets/Scenes/SIBUCAO_AFTERMATH.unity")throw new Exception("Expected aftermath Edit Mode");
var root=GameObject.Find("Aftermath - Phase 3 damaged homes");if(!root||root.transform.childCount!=8)throw new Exception("Missing house groups");
int triangles=0;foreach(var f in root.GetComponentsInChildren<MeshFilter>()){var m=f.sharedMesh;if(!m||m.vertices.Any(p=>!float.IsFinite(p.x)||!float.IsFinite(p.y)||!float.IsFinite(p.z))||m.triangles.Any(i=>i<0||i>=m.vertexCount))throw new Exception("Invalid geometry "+f.name);if(!AssetDatabase.GetAssetPath(m).StartsWith("Assets/StormSim/Sibucao/Art/Aftermath/Phase3/"))throw new Exception("Mesh isolation");var mr=f.GetComponent<Renderer>();foreach(var mat in mr.sharedMaterials)if(!mat||ShaderUtil.ShaderHasError(mat.shader)||!AssetDatabase.GetAssetPath(mat).Contains("/Aftermath/Phase3/"))throw new Exception("Material problem");triangles+=m.triangles.Length/3;}
if(root.GetComponentsInChildren<Collider>().Length!=0||root.GetComponentsInChildren<MonoBehaviour>().Length!=0)throw new Exception("Unexpected runtime component");
if(File.ReadAllText(Dir+"/colliders-before.txt")!=File.ReadAllText(Dir+"/colliders-after.txt"))throw new Exception("Collider records differ");
var roofs=UnityEngine.Object.FindObjectsByType<MeshFilter>(FindObjectsSortMode.None).Where(f=>f.name=="metal"&&f.transform.parent&&(f.transform.parent.name=="Neighbourhood -1_6"||f.transform.parent.name=="Neighbourhood -1_7")).ToArray();foreach(var f in roofs)if(f.GetComponent<MeshCollider>()?.sharedMesh!=f.sharedMesh)throw new Exception("Roof collision mismatch");
var session=UnityEngine.Object.FindFirstObjectByType<SibucaoSession>();var old=GameObject.Find("Aftermath - damaged homes and yards");var terrain=UnityEngine.Object.FindObjectsByType<Collider>(FindObjectsSortMode.None).Where(c=>c.name.StartsWith("Measured terrain")).ToArray();
var names=new[]{"House 1446","House 1511","House 1420","House 1559","House 1572","House 1589","House 1579","House 1608"};var all=UnityEngine.Object.FindObjectsByType<Transform>(FindObjectsInactive.Include,FindObjectsSortMode.None);
var manifest=new List<string>{"name,x,y,z,look_x,look_y,look_z"};
foreach(var name in names){var house=all.Single(t=>t.name==name);var yard=old.GetComponentsInChildren<Transform>().Where(t=>t.name==name+" snapped fence picket").Select(t=>t.position).Aggregate(Vector3.zero,(a,b)=>a+b)/6;var road=session.route.Sample(session.route.Progress(house.position));var toward=road-house.position;toward.y=0;toward.Normalize();var side=Vector3.Cross(Vector3.up,toward);var pos=yard+toward*5+side*1.5f;float y=float.NegativeInfinity;foreach(var t in terrain)if(t.Raycast(new Ray(new Vector3(pos.x,30,pos.z),Vector3.down),out var hit,70))y=Mathf.Max(y,hit.point.y);pos.y=y+1.7f;var target=yard+Vector3.up*1.5f;if(name=="House 1511"||name=="House 1559"||name=="House 1608"){bool found=false;foreach(float distance in new[]{4.0f,3.0f,5.5f}){foreach(float angle in new[]{-30f,30f,-60f,60f,0f}){var candidate=yard+Quaternion.Euler(0,angle,0)*toward*distance;float cy=float.NegativeInfinity;foreach(var tc in terrain)if(tc.Raycast(new Ray(new Vector3(candidate.x,25,candidate.z),Vector3.down),out var ch,60))cy=Mathf.Max(cy,ch.point.y);candidate.y=cy+1.7f;var delta=target-candidate;bool blocked=Physics.SphereCastAll(candidate,.16f,delta.normalized,delta.magnitude*.95f,~0,QueryTriggerInteraction.Ignore).Any(h=>!(h.collider is CharacterController)&&!h.collider.GetComponentInParent<SibucaoResident>());if(!blocked){pos=candidate;found=true;break;}}if(found)break;}}Capture(name.Replace(" ","-")+"-player",pos,target,result);manifest.Add(name+","+pos.x+","+pos.y+","+pos.z+","+target.x+","+target.y+","+target.z);}
Capture("after-roofs-south",new Vector3(-28,16,677),new Vector3(-44,1,660),result);
Capture("after-roofs-north",new Vector3(-52,10,728),new Vector3(-44,2,697),result);
File.WriteAllLines(Dir+"/camera-manifest.csv",manifest);
var lines=new[]{"PASS: 8 house groups; 2 minor, 3 moderate, 3 severe damage sites.","PASS: isolated assets; finite vertices, valid indices, and shaders without reported errors.","PASS: unchanged existing collider snapshots; roof mesh/collider assignments match. Phase 3 adds zero colliders or runtime scripts.","New renderers="+root.GetComponentsInChildren<Renderer>().Length+" triangles="+triangles,"Captured 8 player-height house views and 2 matching roof overviews.","Static visual/collision verification only. Full Play Mode and performance acceptance remains in Phase 7."};
File.WriteAllLines(Dir+"/verification.txt",lines);EditorSceneManager.SaveScene(SceneManager.GetActiveScene());result.Log(string.Join("\n",lines));}
void Capture(string name,Vector3 pos,Vector3 target,ExecutionResult result){var g=new GameObject("Temporary phase 3 review camera");g.hideFlags=HideFlags.HideAndDontSave;result.RegisterObjectCreation(g);var c=g.AddComponent<Camera>();c.transform.position=pos;c.transform.LookAt(target);c.fieldOfView=65;c.farClipPlane=1500;var rt=new RenderTexture(1600,900,24);var t=new Texture2D(1600,900,TextureFormat.RGB24,false);var previous=RenderTexture.active;try{c.targetTexture=rt;c.Render();RenderTexture.active=rt;t.ReadPixels(new Rect(0,0,1600,900),0,0);t.Apply();File.WriteAllBytes(Dir+"/"+name+".png",t.EncodeToPNG());}finally{RenderTexture.active=previous;c.targetTexture=null;rt.Release();result.DestroyObject(t);result.DestroyObject(rt);result.DestroyObject(g);}}}
