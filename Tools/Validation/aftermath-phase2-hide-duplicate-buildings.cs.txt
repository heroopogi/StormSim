using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.SceneManagement;using System;using System.IO;
internal class CommandScript:IRunCommand{public void Execute(ExecutionResult r){if(EditorApplication.isPlaying||SceneManager.GetActiveScene().path!="Assets/Scenes/SIBUCAO_AFTERMATH.unity")throw new Exception("Expected aftermath Edit Mode");foreach(var t in Resources.FindObjectsOfTypeAll<MonoBehaviour>()){if(!t.gameObject.scene.IsValid()||t.GetType().Name!="Cesium3DTileset"||t.name!="Cesium OSM Buildings")continue;var p=t.GetType().GetProperty("ionAssetID");if(Convert.ToInt64(p.GetValue(t))!=96188)throw new Exception("Unexpected tileset");r.RegisterObjectModification(t.gameObject);t.gameObject.SetActive(false);r.Log("Disabled duplicate Cesium OSM Buildings tileset 96188 in aftermath scene; custom buildings retained.");}EditorSceneManager.SaveScene(SceneManager.GetActiveScene());}}
